<script>
function hideAll(){
  document.getElementById('home').style.display='none';
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
}
function goHome(){
  document.getElementById('home').style.display='block';
  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));
  window.scrollTo({top:0,behavior:'smooth'});
}
function openMayor(){
  hideAll();
  document.getElementById('mayor').classList.add('active');
  document.getElementById('followTool').style.display='none';
  document.getElementById('registerTool').style.display='none';
  document.getElementById('status').textContent='';
  window.scrollTo({top:0,behavior:'smooth'});
}
function showTool(which){
  document.getElementById('followTool').style.display=which==='follow'?'block':'none';
  document.getElementById('registerTool').style.display=which==='register'?'block':'none';
}
function openPage(name){
  hideAll();
  document.getElementById('otherTitle').textContent=name;
  document.getElementById('other').classList.add('active');
  window.scrollTo({top:0,behavior:'smooth'});
}
document.getElementById('followForm').addEventListener('submit',function(e){
  e.preventDefault();
  const data=new FormData(this);
  const card=document.createElement('article');
  card.className='list-card';
  const title=document.createElement('h3');
  title.textContent=data.get('person')+' · '+data.get('kind');
  const note=document.createElement('p');
  note.textContent=data.get('note')||'پیگیری ثبت شد';
  card.append(title,note);
  const tag=document.createElement('span');
  tag.className='pill gold';
  tag.textContent=data.get('date')||'پیگیری جدید';
  card.append(tag);
  document.getElementById('followList').prepend(card);
  document.getElementById('status').textContent='پیگیری به فهرست همین صفحه اضافه شد؛ هنوز دائمی ذخیره نشده است.';
  this.reset();
});
document.getElementById('fileForm').addEventListener('submit',function(e){
  e.preventDefault();
  const data=new FormData(this);
  const card=document.createElement('article');
  card.className='list-card';
  const title=document.createElement('h3');
  title.textContent=data.get('deal')+' · '+data.get('area');
  const detail=document.createElement('p');
  detail.textContent=(data.get('size')||'—')+' متر · '+(data.get('rooms')||'—')+' اتاق · '+(data.get('price')||'قیمت اعلام نشده');
  card.append(title,detail);
  const tag=document.createElement('span');
  tag.className='pill '+(data.get('status')==='فوری'?'red':'gold');
  tag.textContent=data.get('status');
  card.append(tag);
  document.getElementById('urgentList').prepend(card);
  if(data.get('status')==='فوری'){
    const counter=document.getElementById('urgentNumber');
    counter.textContent=Number(counter.textContent)+1;
  }
  document.getElementById('status').textContent='فایل به فهرست همین صفحه اضافه شد؛ هنوز دائمی ذخیره نشده است.';
  this.reset();
});
